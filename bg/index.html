  - resource: "https://spotpuls.app/api/v1/prices?zone=BG&vat=1"
    scan_interval: 900
    sensor:
      - name: "Spotpuls BG Current Spot"
        value_template: "{{ value_json.current.spotPrice }}"
        unit_of_measurement: "ст./kWh"
        json_attributes_path: "$.summary"
        json_attributes:
          - todayMin
          - todayMax
          - rolling30dAvg
