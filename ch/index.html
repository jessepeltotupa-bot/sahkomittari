HOME ASSISTANT (CONFIGURATION.YAML) 
rest:
  - resource: "https://spotpuls.app/api/v1/prices?zone=CH&vat=1"
    scan_interval: 900
    sensor:
      - name: "Spotpuls CH Current Spot"
        value_template: "{{ value_json.current.spotPrice }}"
        unit_of_measurement: "Rp./kWh"
        json_attributes_path: "$.summary"
        json_attributes:
          - todayMin
          - todayMax
          - rolling30dAvg
