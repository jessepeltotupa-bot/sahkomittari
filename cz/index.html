HOME ASSISTANT (CONFIGURATION.YAML) 
rest:
  - resource: "https://spotpuls.app/api/v1/prices?zone=CZ&vat=1"
    scan_interval: 900
    sensor:
      - name: "Spotpuls CZ Current Spot"
        value_template: "{{ value_json.current.spotPrice }}"
        unit_of_measurement: "Kč/kWh"
        json_attributes_path: "$.summary"
        json_attributes:
          - todayMin
          - todayMax
          - rolling30dAvg
