HOME ASSISTANT (CONFIGURATION.YAML) 
rest:
  - resource: "https://spotpuls.app/api/v1/prices?zone=DK2&vat=1"
    scan_interval: 900
    sensor:
      - name: "Spotpuls DK2 Current Spot"
        value_template: "{{ value_json.current.spotPrice }}"
        unit_of_measurement: "øre/kWh"
        json_attributes_path: "$.summary"
        json_attributes:
          - todayMin
          - todayMax
          - rolling30dAvg
