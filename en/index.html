rest:
  - resource: "https://spotpuls.app/api/v1/prices?zone=DE-LU&vat=1"
    scan_interval: 900
    sensor:
      - name: "Spotpuls DE-LU Current Spot"
        value_template: "{{ value_json.current.spotPrice }}"
        unit_of_measurement: "ct/kWh"
        json_attributes_path: "$.summary"
        json_attributes:
          - todayMin
          - todayMax
          - rolling30dAvg
