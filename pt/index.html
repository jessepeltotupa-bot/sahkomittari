rest:
  - resource: "https://spotpuls.app/api/v1/prices?zone=PT&vat=1"

      - name: "Spotpuls PT Current Spot"
        value_template: "{{ value_json.current.spotPrice }}"
        unit_of_measurement: "cênt./kWh"
        json_attributes_path: "$.summary"
        json_attributes:
          - todayMin
          - todayMax
          - rolling30dAvg
