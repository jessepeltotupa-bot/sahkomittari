HOME ASSISTANT (CONFIGURATION.YAML) 
rest:
  - resource: "https://spotpuls.app/api/v1/prices?zone=SE3&vat=1"
    scan_interval: 900
    sensor:
      - name: "Spotpuls SE3 Current Spot"
        value_template: "{{ value_json.current.spotPrice }}"
        unit_of_measurement: "öre/kWh"
        json_attributes_path: "$.summary"
        json_attributes:
          - todayMin
          - todayMax
          - rolling30dAvg
